3) COMO GERAR O .apk (comandos exatos usados)
------------------------------------------------------------
# Pré-requisitos (1x)
sudo apt-get update && sudo apt-get install -y openjdk-17-jdk-headless
export ANDROID_HOME=/home/user/android-sdk
export ANDROID_SDK_ROOT=$ANDROID_HOME
export PATH=$ANDROID_HOME/cmdline-tools/latest/bin:$PATH

# Baixar SDK tools + plataforma + build-tools (1x)
cd /home/user/android-sdk/cmdline-tools
curl -fsSL -o clt.zip https://dl.google.com/android/repository/commandlinetools-linux-11076708_latest.zip
unzip -q clt.zip && mv cmdline-tools latest && rm clt.zip
yes | sdkmanager --licenses
sdkmanager --install "platform-tools" "platforms;android-34" "build-tools;34.0.0"

export BT=$ANDROID_HOME/build-tools/34.0.0
export PLATFORM=$ANDROID_HOME/platforms/android-34/android.jar

# Build (a cada compilação)
cd /home/user/apkbuild && PROJ=Cine-2024-APK
mkdir -p out/res out/gen out/classes
$BT/aapt2 compile --dir $PROJ/app/src/main/res -o out/res
$BT/aapt2 link -o out/app-unsigned.apk -I $PLATFORM \
  --manifest $PROJ/app/src/main/AndroidManifest.xml \
  -A $PROJ/app/src/main/assets --java out/gen \
  $(find out/res -name '*.flat') \
  --min-sdk-version 21 --target-sdk-version 34
find $PROJ/app/src/main/java out/gen -name '*.java' > out/sources.txt
javac -encoding UTF-8 -source 8 -target 8 -classpath $PLATFORM -d out/classes @out/sources.txt
jar cf out/classes.jar -C out/classes .
$BT/d8 --release --lib $PLATFORM --output out/ out/classes.jar
cp out/app-unsigned.apk out/app-withdex.apk
cd out && zip -q app-withdex.apk classes.dex && cd ..
$BT/zipalign -f 4 out/app-withdex.apk out/app-aligned.apk
keytool -genkeypair -keystore /home/user/apkbuild/cinemax.keystore \
  -alias cinemax -keyalg RSA -keysize 2048 -validity 10000 \
  -storepass cinemax123 -keypass cinemax123 \
  -dname "CN=CineMax TV, OU=CineMax, O=CineMax, L=BR, ST=BR, C=BR"
$BT/apksigner sign --ks /home/user/apkbuild/cinemax.keystore \
  --ks-pass pass:cinemax123 --ks-key-alias cinemax --key-pass pass:cinemax123 \
  --out out/CineMaxTV.apk out/app-aligned.apk
# Publicar: copy para /tmp/creao-artifacts e usar publish_artifact.

OBS: se o manifest não tiver package, aapt2 exige
  package="br.com.cinemax.tv" no <manifest>.
